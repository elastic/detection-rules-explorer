About

Detects a user agent that identifies an LLM-driven penetration testing framework. ARTEX, its AntiFox and ScopeWeaver forks, CyberStrikeAI, and HexStrike embed these strings in web requests.

Query
✄𐘗
text code block:
✄𐘗from
    logs-aws.cloudtrail-*,
    logs-aws.eks_audit-*,
    logs-nginx.access-*,
    logs-apache.access-*,
    logs-apache_tomcat.access-*,
    logs-iis.access-*,
    logs-traefik.access-*,
    logs-zeek.http*,
    logs-network_traffic.http*,
    logs-azure.signinlogs-*,
    logs-azure.graphactivitylogs-*,
    logs-azure.aadgraphactivitylogs-*,
    logs-o365.audit-*,
    logs-kubernetes.audit_logs-*,
    logs-gcp.audit-*,
    logs-gcp_vertexai.auditlogs-*,
    logs-anthropic.audit-*,
    logs-openai_chatgpt_enterprise.auth_log-*,
    logs-okta.system-*
    metadata _id, _version, _index
| where user_agent.original is not null
| eval Esql.ua_lower = to_lower(user_agent.original)
| eval Esql.framework = case(
    Esql.ua_lower like "artex-*" or Esql.ua_lower like "mozilla/* artex*" or Esql.ua_lower like "norma/0.4*", "ARTEX",
    Esql.ua_lower like "*antifox*", "AntiFox",
    Esql.ua_lower like "*scopeweaver*", "ScopeWeaver",
    Esql.ua_lower like "*cyberstrikeai*", "CyberStrikeAI",
    Esql.ua_lower like "*hexstrike-http-framework*" or Esql.ua_lower like "*hexstrike-browseragent*", "HexStrike",
    "other"
)
| where Esql.framework != "other"
| keep
    _id,
    _version,
    _index,
    @timestamp,
    Esql.framework,
    Esql.ua_lower,
    user_agent.original,
    source.ip,
    data_stream.dataset,
    data_stream.namespace,
    host.name
