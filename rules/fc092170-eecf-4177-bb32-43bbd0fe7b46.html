Elastic Security Detection Rules

About

Hunts for HTTP artifacts associated with exploitation of CVE-2026-20079 against Cisco Secure Firewall Management Center (FMC). The queries identify remote use of the boot-created `csm_processes` web session, a `validateLicense` request that writes attacker-controlled script content, and an `upgradeReadinessCall` request referencing `license.tmp`. These artifacts are based on independently reproduced public exploit mechanics and should be investigated as attempted or successful exploitation of the FMC management interface.

✄𐘗FROM logs-network_traffic.http-* METADATA _id, _index
| WHERE @timestamp > NOW() - 30 days
| WHERE url.path IN ("/help/about.cgi", "/login.cgi", "/ui/user/general", "/sajaxintf.cgi", "/pjb.cgi")
    AND request LIKE "*CGISESSID=csm_processes*"
| KEEP
    @timestamp,
    source.ip,
    source.port,
    destination.ip,
    destination.port,
    http.request.method,
    url.path,
    url.query,
    http.response.status_code,
    user_agent.original,
    request,
    _index,
    _id
| SORT @timestamp DESC
| LIMIT 1000
FROM logs-network_traffic.http-*, logs-modsecurity.auditlog-* METADATA _id, _index
| WHERE @timestamp > NOW() - 30 days
| WHERE http.request.method == "POST"
    AND url.path == "/sajaxintf.cgi"
    AND http.request.body.content LIKE "*validateLicense*"
    AND (
        http.request.body.content LIKE "*This script was generated using Makeself*"
        OR http.request.body.content LIKE "*#!/bin/sh*"
    )
| KEEP
    @timestamp,
    source.ip,
    source.port,
    destination.ip,
    destination.port,
    http.request.method,
    url.path,
    url.query,
    http.request.mime_type,
    http.response.status_code,
    user_agent.original,
    http.request.body.content,
    _index,
    _id
| SORT @timestamp DESC
| LIMIT 1000
FROM logs-network_traffic.http-*, logs-modsecurity.auditlog-* METADATA _id, _index
| WHERE @timestamp > NOW() - 30 days
| WHERE http.request.method == "POST"
    AND url.path == "/pjb.cgi"
    AND http.request.body.content LIKE "*upgradeReadinessCall*"
    AND http.request.body.content LIKE "*license.tmp*"
| KEEP
    @timestamp,
    source.ip,
    source.port,
    destination.ip,
    destination.port,
    http.request.method,
    url.path,
    url.query,
    http.request.mime_type,
    http.response.status_code,
    user_agent.original,
    http.request.body.content,
    _index,
    _id
| SORT @timestamp DESC
| LIMIT 1000
